# 第一部分：基础数据结构

## 模块 1：数组与双指针 / 滑动窗口

### 核心概念
数组是最基础的线性结构，支持 O(1) 随机访问。围绕数组最常见的两类技巧是**双指针**（首尾指针相向而行，或快慢指针同向而行）和**滑动窗口**（维护一个动态区间）。

### 手写模板

```python
# 模板 1：相向双指针
def two_pointer_template(arr):
    left, right = 0, len(arr) - 1
    while left < right:
        # 根据条件移动指针
        if condition(arr[left], arr[right]):
            left += 1
        else:
            right -= 1

# 模板 2：滑动窗口（求满足条件的最长/最短子数组）
def sliding_window_template(arr, target):
    left = 0
    window_sum = 0
    best = 0
    for right in range(len(arr)):
        window_sum += arr[right]
        while window_sum > target:   # 窗口不满足条件时收缩左边界
            window_sum -= arr[left]
            left += 1
        best = max(best, right - left + 1)
    return best
```

```python
def three_sum(nums):
    nums.sort()
    res = []
    n = len(nums)
    for i in range(n - 2):
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        left, right = i + 1, n - 1
        while left < right:
            s = nums[i] + nums[left] + nums[right]
            if s == 0:
                res.append([nums[i], nums[left], nums[right]])
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1; right -= 1
            elif s < 0:
                left += 1
            else:
                right -= 1
    return res
```

```python
from collections import Counter

def min_window(s, t):
    need = Counter(t)
    missing = len(t)
    left = start = end = 0
    for right, ch in enumerate(s, 1):
        if need[ch] > 0:
            missing -= 1
        need[ch] -= 1
        if missing == 0:
            while left < right and need[s[left]] < 0:
                need[s[left]] += 1
                left += 1
            if end == 0 or right - left < end - start:
                start, end = left, right
            need[s[left]] += 1
            missing += 1
            left += 1
    return s[start:end]
```

## 模块 3：栈与队列

### 核心概念
栈是后进先出（LIFO），常用于括号匹配、单调栈、表达式求值；队列是先进先出（FIFO），常用于 BFS、任务调度。**单调栈**是高频考点：维护一个单调递增/递减的栈来快速找到"下一个更大/更小元素"。

### 手写模板

```python
# 模板 1：单调栈（找右边第一个更大的元素）
def next_greater_element(nums):
    stack = []           # 存下标
    res = [-1] * len(nums)
    for i, num in enumerate(nums):
        while stack and nums[stack[-1]] < num:
            res[stack.pop()] = num
        stack.append(i)
    return res

# 模板 2：用两个栈实现队列
class MyQueue:
    def __init__(self):
        self.in_stack, self.out_stack = [], []

    def push(self, x):
        self.in_stack.append(x)

    def pop(self):
        self._transfer()
        return self.out_stack.pop()

    def _transfer(self):
        if not self.out_stack:
            while self.in_stack:
                self.out_stack.append(self.in_stack.pop())
```

## 模块 4：哈希表

### 核心概念
哈希表通过哈希函数将键映射到桶，平均 O(1) 的插入、删除、查找。核心应用：计数、去重、映射关系、前缀和优化。

### 手写模板

```python
# 模板：手写简易哈希表（拉链法解决冲突）
class MyHashMap:
    def __init__(self, size=1000):
        self.size = size
        self.buckets = [[] for _ in range(size)]

    def _hash(self, key):
        return hash(key) % self.size

    def put(self, key, value):
        idx = self._hash(key)
        for i, (k, v) in enumerate(self.buckets[idx]):
            if k == key:
                self.buckets[idx][i] = (key, value)
                return
        self.buckets[idx].append((key, value))

    def get(self, key):
        idx = self._hash(key)
        for k, v in self.buckets[idx]:
            if k == key:
                return v
        return -1

    def remove(self, key):
        idx = self._hash(key)
        self.buckets[idx] = [(k, v) for k, v in self.buckets[idx] if k != key]
```

```python
        self.cache = OrderedDict()
        self.capacity = capacity

            return -1
```

## 模块 5：二叉树与二叉搜索树

### 核心概念
树是层次结构，二叉搜索树（BST）左子树 < 根 < 右子树，中序遍历为有序序列。核心技巧：**递归三要素**（终止条件、单层逻辑、返回值）、**前中后序/层序遍历**。

### 手写模板

```python
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

# 模板 1：递归遍历（前序示例，改一下顺序即可变中序/后序）
def preorder(root):
    if not root:
        return []
    return [root.val] + preorder(root.left) + preorder(root.right)

# 模板 2：层序遍历（BFS）
from collections import deque

def level_order(root):
    if not root:
        return []
    res, queue = [], deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left: queue.append(node.left)
            if node.right: queue.append(node.right)
        res.append(level)
    return res
```

## 模块 6：堆（优先队列）

### 核心概念
堆是一种完全二叉树，最大堆的根节点是最大值，最小堆的根节点是最小值。常用于 **Top-K 问题**、**动态求最值**、**任务调度**。Python 用 `heapq` 实现最小堆（若需要最大堆，取负数存入）。

### 手写模板

```python
import heapq

# 模板 1：Top-K 大的元素（维护一个大小为 K 的最小堆）
def top_k_largest(nums, k):
    heap = nums[:k]
    heapq.heapify(heap)
    for num in nums[k:]:
        if num > heap[0]:
            heapq.heapreplace(heap, num)
    return heap

# 模板 2：手写堆的上浮/下沉操作（理解底层原理）
class MinHeap:
    def __init__(self):
        self.heap = []

    def push(self, val):
        self.heap.append(val)
        self._sift_up(len(self.heap) - 1)

    def pop(self):
        top = self.heap[0]
        last = self.heap.pop()
        if self.heap:
            self.heap[0] = last
            self._sift_down(0)
        return top

    def _sift_up(self, i):
        while i > 0:
            parent = (i - 1) // 2
            if self.heap[parent] <= self.heap[i]:
                break
            self.heap[parent], self.heap[i] = self.heap[i], self.heap[parent]
            i = parent

    def _sift_down(self, i):
        n = len(self.heap)
        while True:
            smallest = i
            left, right = 2 * i + 1, 2 * i + 2
            if left < n and self.heap[left] < self.heap[smallest]:
                smallest = left
            if right < n and self.heap[right] < self.heap[smallest]:
                smallest = right
            if smallest == i:
                break
            self.heap[i], self.heap[smallest] = self.heap[smallest], self.heap[i]
            i = smallest
```

## 模块 7：二分查找

### 核心概念
二分查找不止能在有序数组里找值，还能求解**满足单调性的最优化问题**（例如"最小化最大值"类题目）。关键是想清楚 `left`、`right`、`mid` 的开闭区间定义，避免死循环。

### 手写模板

```python
# 模板 1：查找目标值（左闭右闭区间）
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

# 模板 2：查找左边界 / 右边界
def lower_bound(arr, target):
    left, right = 0, len(arr)
    while left < right:
        mid = (left + right) // 2
        if arr[mid] < target:
            left = mid + 1
        else:
            right = mid
    return left  # 第一个 >= target 的位置

# 模板 3：二分答案（在结果空间上二分）
def binary_search_on_answer(lo, hi, check):
    while lo < hi:
        mid = (lo + hi) // 2
        if check(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo
```

## 模块 9：DFS / BFS（图与网格）

### 核心概念
DFS 用递归/栈"一条路走到黑"，适合搜索所有路径、连通分量；BFS 用队列"逐层扩散"，适合求**最短路径**（无权图/网格）。网格类问题是 DFS/BFS 的高频载体。

### 手写模板

```python
# 模板 1：网格 DFS（岛屿类问题）
def dfs_grid(grid, r, c, visited):
    rows, cols = len(grid), len(grid[0])
    if r < 0 or r >= rows or c < 0 or c >= cols or (r, c) in visited or grid[r][c] == 0:
        return
    visited.add((r, c))
    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
        dfs_grid(grid, r + dr, c + dc, visited)

# 模板 2：网格 BFS（最短路径）
from collections import deque

def bfs_grid(grid, start, end):
    rows, cols = len(grid), len(grid[0])
    queue = deque([(start, 0)])
    visited = {start}
    while queue:
        (r, c), dist = queue.popleft()
        if (r, c) == end:
            return dist
        for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and (nr, nc) not in visited and grid[nr][nc] != 1:
                visited.add((nr, nc))
                queue.append(((nr, nc), dist + 1))
    return -1
```

## 模块 10：回溯算法

### 核心概念
回溯是 DFS 的一种应用，用于枚举所有可能的解（排列、组合、子集），核心是"做选择 → 递归 → 撤销选择"。剪枝是回溯效率的关键。

### 手写模板

```python
def backtrack_template(choices, path, res):
    if is_end_condition(path):        # 终止条件
        res.append(list(path))
        return
    for choice in choices:
        if not is_valid(choice, path):  # 剪枝
            continue
        path.append(choice)             # 做选择
        backtrack_template(choices, path, res)
        path.pop()                      # 撤销选择
```

```python
    res = []
    cols, diag1, diag2 = set(), set(), set()

    def backtrack(row, path):
        if row == n:
            res.append(list(path))
            return
        for col in range(n):
            if col in cols or (row - col) in diag1 or (row + col) in diag2:
```

## 模块 12：动态规划

### 核心概念
动态规划把问题拆成重叠子问题，用状态转移方程避免重复计算。做题的核心是想清楚：**dp 数组定义什么、初始值是什么、状态转移方程是什么、遍历顺序是什么**。常见分类：线性 DP、背包 DP、区间 DP、状态压缩 DP、树形 DP。

### 手写模板

```python
# 模板 1：一维 DP
def dp_1d_template(n):
    dp = [0] * (n + 1)
    dp[0] = BASE_CASE
    for i in range(1, n + 1):
        dp[i] = transition(dp[i - 1], ...)
    return dp[n]

# 模板 2：二维 DP（网格/双字符串）
def dp_2d_template(m, n):
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            dp[i][j] = transition(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]
```

```python
    n = len(dist)
    dp = [[float('inf')] * n for _ in range(1 << n)]
    dp[1][0] = 0  # 从城市 0 出发
    for mask in range(1 << n):
        for u in range(n):

                continue
            for v in range(n):
                if mask & (1 << v):
                    continue
                new_mask = mask | (1 << v)
                dp[new_mask][v] = min(dp[new_mask][v], dp[mask][u] + dist[u][v])
    full_mask = (1 << n) - 1
```

## 模块 13：分治算法

### 核心概念
分治把问题拆成若干独立子问题，分别求解后合并结果，核心是"分解 → 解决 → 合并"，归并排序和快速排序都是经典分治。

### 经典题训练

**1. 最大子数组和（分治版，与 Kadane 对照）**

```python
def max_subarray_divide(nums):
    def helper(left, right):
        if left == right:
            return nums[left]
        mid = (left + right) // 2
        left_max = helper(left, mid)
        right_max = helper(mid + 1, right)
        cross_max = cross_sum(left, mid, right)
        return max(left_max, right_max, cross_max)

    def cross_sum(left, mid, right):
        left_sum = float('-inf'); s = 0
        for i in range(mid, left - 1, -1):
            s += nums[i]; left_sum = max(left_sum, s)
        right_sum = float('-inf'); s = 0
        for i in range(mid + 1, right + 1):
            s += nums[i]; right_sum = max(right_sum, s)
        return left_sum + right_sum

    return helper(0, len(nums) - 1)
```

```python
    count = 0

    def merge_sort(arr):
        nonlocal count
        if len(arr) <= 1:
            return arr
        mid = len(arr) // 2
        left = merge_sort(arr[:mid])
        right = merge_sort(arr[mid:])
        merged, i, j = [], 0, 0
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i]); i += 1
            else:
                count += len(left) - i  # 关键：统计逆序对
                merged.append(right[j]); j += 1
        merged.extend(left[i:]); merged.extend(right[j:])
        return merged

    merge_sort(nums)
    return count
```

```python
    for i in range(n):
        for j in range(i + 1, n):
            if is_connected[i][j] == 1:
                union(i, j)
    return len({find(i) for i in range(n)})
```

**2. 冗余连接（LeetCode 684，找出使图成环的边）**
```python
def find_redundant_connection(edges):
    n = len(edges)
    parent = list(range(n + 1))

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    for u, v in edges:
        ru, rv = find(u), find(v)
        if ru == rv:
            return [u, v]
        parent[ru] = rv
    return []
```

**3. 账户合并（LeetCode 721，并查集 + 哈希表）**
```python
from collections import defaultdict

def accounts_merge(accounts):
    parent = list(range(len(accounts)))

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        parent[find(x)] = find(y)

    email_to_idx = {}
    for i, account in enumerate(accounts):
        for email in account[1:]:
            if email in email_to_idx:
                union(i, email_to_idx[email])
            else:
                email_to_idx[email] = i

    groups = defaultdict(set)
    for email, idx in email_to_idx.items():
        groups[find(idx)].add(email)

    return [[accounts[root][0]] + sorted(emails) for root, emails in groups.items()]
```

## 模块 15：字符串匹配与 Trie

### 核心概念
除 KMP 外，**Trie（字典树）** 是处理字符串前缀查找的核心结构，插入/查询复杂度为 O(单词长度)，广泛用于自动补全、单词搜索。

### 手写模板

```python
class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word):
        node = self.root
        for ch in word:
            node = node.children.setdefault(ch, TrieNode())
        node.is_end = True

    def search(self, word):
        node = self._find(word)
        return node is not None and node.is_end

    def starts_with(self, prefix):
        return self._find(prefix) is not None

    def _find(self, word):
        node = self.root
        for ch in word:
            if ch not in node.children:
                return None
            node = node.children[ch]
        return node
```

```python
class WordDictionary:
    def __init__(self):
        self.root = TrieNode()

    def add_word(self, word):
        node = self.root
        for ch in word:
            node = node.children.setdefault(ch, TrieNode())
        node.is_end = True

    def search(self, word):
        def dfs(node, i):
            if i == len(word):
                return node.is_end
            ch = word[i]
            if ch == '.':
                return any(dfs(child, i + 1) for child in node.children.values())
            if ch not in node.children:
                return False
            return dfs(node.children[ch], i + 1)

        return dfs(self.root, 0)
```
